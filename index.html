-- شغّليه مرة واحدة في Supabase → SQL Editor -- الهدف: كل طالبة (مسجّلة الدخول) ترى صفّها فقط، لا صف "محمد" ولا صفوف غيرها. -- ملاحظة: المعلمات والإدارة يعملن بالمفتاح العام (anon) فتبقى صلاحياتهن كما هي. -- 0) (اختياري) شاهدي السياسات الحالية قبل التغيير: -- select tablename, policyname, roles, cmd, qual from pg_policies -- where schemaname='public' and tablename in ('students','progress','reports','notifications','teachers'); -- 1) دالة توحّد صيغة الرقم (نفس منطق norm في التطبيق) create or replace function public.norm_phone(p text) returns text language sql immutable as $$ select ltrim(regexp_replace( translate(coalesce(p,''), '٠١٢٣٤٥٦٧٨٩', '0123456789'), '\D', '', 'g'), '0') $$; -- 2) رقم الطالبة الحالية من بريدها في حساب الدخول create or replace function public.my_phone() returns text language sql stable as $$ select public.norm_phone(split_part(coalesce(auth.jwt()->>'email',''), '@', 1)) $$; -- 3) تفعيل RLS وحذف السياسات القديمة على هذه الجداول ثم إعادة بنائها do $$ declare t text; pol record; begin foreach t in array array['students','progress','reports','notifications','teachers'] loop execute format('alter table public.%I enable row level security', t); for pol in select policyname from pg_policies where schemaname='public' and tablename=t loop execute format('drop policy %I on public.%I', pol.policyname, t); end loop; -- المعلمات/الإدارة (مفتاح anon) كما كان يعمل التطبيق execute format('create policy rattil_anon_all on public.%I for all to anon using (true) with check (true)', t); end loop; end $$; -- 4) الطالبة: ترى صفّها فقط create policy rattil_student_own on public.students for select to authenticated using (public.norm_phone(phone) = public.my_phone()); create policy rattil_student_progress on public.progress for select to authenticated using (student_id in (select id from public.students where public.norm_phone(phone) = public.my_phone())); create policy rattil_student_reports on public.reports for select to authenticated using (student_id in (select id from public.students where public.norm_phone(phone) = public.my_phone())); create policy rattil_student_notifications on public.notifications for select to authenticated using (student_id in (select id from public.students where public.norm_phone(phone) = public.my_phone())); -- الطالبة تحتاج اسم معلمتها ورقمها فقط create policy rattil_student_teachers on public.teachers for select to authenticated using (true); -- 5) تنبيه مهم: الدالتان check_my_expiry_notice و mark_my_notifications_read -- يجب أن تكونا SECURITY DEFINER وتعتمدان على my_phone() أو auth.uid() — راجعيهما إن لم تعملا.
